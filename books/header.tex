\def\countproblems{%
  \setcounter{pcount}{0}%
  \def\halfgoban{}%

\def\rendertitle{
  \newgeometry{margin=100pt, top=230pt}%
  \thispagestyle{empty}
  \raggedright
  \HUGE\entitle\phantom{g}\\
  \vspace{5pt}
  \Huge
  \ifdefined\jptitle\begin{CJK}{UTF8}{min}{\jptitle}\end{CJK}\fi%
  \ifdefined\kotitle\begin{CJK}{UTF8}{mj}{\kotitle}\end{CJK}\fi%
  \ifdefined\zhtitle\begin{CJK}{UTF8}{bsmi}{\zhtitle}\end{CJK}\fi%
  \vspace{30pt}
  \Large
  \par Problems: \countproblems
  \par Difficulty: \level
  \par Source: \href{\source}{\source}
  \par All problems are black to play
  \clearpage
}

\def\renderproblems{%
  \setcounter{pcount}{0}%
  \fancyhfoffset{-20pt}%
  \newgeometry{margin=40pt, top=102pt, bottom=80pt, headsep=40pt}%
  \ifdefined\solutionbook
    \def\line##1{##1\\}%
    \setbox2=\hbox{{\small\gnos +++++++++++++++++++}}%
    \gnosBoardWidthDim=\wd2%
  \else
    \def\line##1{\truncate[]{150pt}{##1}\\}%
  \fi
  \def\hvspace{%
    \hspace{5pt}%
    \vspace{5pt}%
  }%
  \def\halfgoban{%
    \def\line####1{####1\\}%
    \def\hvspace{%
      \hspace{18.2pt}%
      \vspace{7.5pt}%
    }%
    \clearpage%
    \newgeometry{margin=40pt, top=102pt, bottom=10pt, headsep=40pt}%
    \hspace{-18.2pt}%
  }%
  \def\halfgobanphantom{
    \def\line####1{####1\\}%
    \def\hvspace{%
      \hspace{18.2pt}%
      \vspace{7.5pt}%
    }%
    \hspace{-18.2pt}%
  }%
  \def\fullgoban{%
    \def\line####1{####1\\}%
    \def\hvspace{%
      \hspace{18.2pt}%
      \vspace{7.5pt}%
    }%
    \clearpage%
    \newgeometry{margin=40pt, top=102pt, bottom=10pt, headsep=40pt}%
    \hspace{-18.2pt}%
  }%
  \def\fullgobanphantom{%
    \def\line####1{####1\\}%
    \def\hvspace{%
      \hspace{18.2pt}%
      \vspace{7.5pt}%
    }%
    \hspace{-18.2pt}%
  }%
  \ifdefined\solutionbook
    \let\halfgoban\relax
    \let\halfgobanphantom\relax
    \let\fullgoban\relax
    \let\fullgobanphantom\relax
  \fi
  \def\pfull##1##2{%
    \stepcounter{pcount}%
    \hvspace%
    \begin{varwidth}{\textwidth}%
      \center%
      \small%
      \def\line####1{####1\\}%
      \let\pfullorig\gnos%
      \def\gnos{\pfullorig%
        \line{<(((((((((((((((((>}%
        \line{\char91+++++++++++++++++]}%
        \line{\char91+++++++++++++++++]}%
        \line{\char91++*+++++*+++++*++]}%
        \line{\char91+++++++++++++++++]}%
        \line{\char91+++++++++++++++++]}%
        \line{\char91+++++++++++++++++]}%
        \line{\char91+++++++++++++++++]}%
      }%
      \input{problems/\bookdir/##1/##2.\gnosext}%
      \normalsize\plabel{##1}{##2}%
      \gnosnotes{##1}{##2}%
    \end{varwidth}%
    \allowbreak%
    \hfil%
  }%
  \def\p##1##2{%
    \stepcounter{pcount}%
    \hvspace%
    \begin{varwidth}{\textwidth}%
      \center%
      \small%
      \input{problems/\bookdir/##1/##2.\gnosext}%
      \normalsize\plabel{##1}{##2}%
      \gnosnotes{##1}{##2}%
    \end{varwidth}%
    \allowbreak%
    \hfil%
  }%
  \begin{center}%
  \hspace{-5pt}%
  \problems%
  \ifdefined\solutionbook\hspace{\gnosBoardWidthDim}\hfil\fi%
  \end{center}%
  \clearpage%
}

\def\rendersolutions{%
  \setcounter{pcount}{0}%
  \fancyhfoffset{0pt}%
  \newgeometry{margin=60pt, top=82pt, bottom=80pt}%
  \def\halfgoban{}%
  \def\halfgobanphantom{}%
  \def\fullgoban{}%
  \def\fullgobanphantom{}%
  \def\pfull##1##2{%
    \stepcounter{pcount}%
    {%
      (\href{https://www.101weiqi.com/q/##2}{\thepcount})%
      \:\:%
      \input{problems/\bookdir/##1/##2.solution}%
    }

  }%
  \def\p##1##2{%
    \stepcounter{pcount}%
    {%
      (\href{https://www.101weiqi.com/q/##2}{\thepcount})%
      \:\:%
      \input{problems/\bookdir/##1/##2.solution}%
    }

  }%
  \begin{multicols*}{2}%
    \setlength{\parskip}{2.7pt}%
    \footnotesize%
    \problems%
  \end{multicols*}%
  \clearpage%
}
